* **Implementation Tasks**:
  1. **Hard Negative Mining Contrastive Loss** in `src/hard_neg_loss.py`:
     ```python
     import torch
     import torch.nn as nn
     import torch.nn.functional as F

     class HardNegativeContrastiveLoss(nn.Module):
         def __init__(self, temperature: float = 0.05, top_k_hard: int = 5):
             super().__init__()
             self.temperature = temperature
             self.top_k_hard = top_k_hard

         def forward(self, query_embs: torch.Tensor, pos_embs: torch.Tensor,
                     neg_pool: torch.Tensor) -> torch.Tensor:
             # query_embs: [B, D], pos_embs: [B, D], neg_pool: [B, num_negs, D]
             q_norm = F.normalize(query_embs, p=2, dim=-1)
             p_norm = F.normalize(pos_embs, p=2, dim=-1)
             n_norm = F.normalize(neg_pool, p=2, dim=-1)

             # Positive similarity: [B, 1]
             pos_sim = torch.sum(q_norm * p_norm, dim=-1, keepdim=True) / self.temperature

             # Negative similarity: [B, num_negs]
             neg_sim = torch.einsum("b d, b n d -> b n", q_norm, n_norm) / self.temperature

             # Mine top-k hardest negatives (highest similarity)
             hardest_negs, _ = torch.topk(neg_sim, k=self.top_k_hard, dim=-1)

             # Combined logits: [B, 1 + top_k_hard]
             logits = torch.cat([pos_sim, hardest_negs], dim=-1)
             labels = torch.zeros(query_embs.shape[0], dtype=torch.long, device=query_embs.device)

             return F.cross_entropy(logits, labels)
     ```
