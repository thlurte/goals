     ```python
     import torch
     import torch.nn as nn
     import torch.nn.functional as F

     class MarginMSEDistillationLoss(nn.Module):
         def __init__(self):
             super().__init__()

         def forward(self,
                     student_pos_score: torch.Tensor, # [B]
                     student_neg_score: torch.Tensor, # [B]
                     teacher_pos_score: torch.Tensor, # [B]
                     teacher_neg_score: torch.Tensor  # [B]
         ) -> torch.Tensor:
             student_margin = student_pos_score - student_neg_score
             teacher_margin = teacher_pos_score - teacher_neg_score
             return F.mse_loss(student_margin, teacher_margin)
     ```
