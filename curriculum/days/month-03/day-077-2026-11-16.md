     ```python
     import matplotlib.pyplot as plt
     import numpy as np
     import subprocess
     import json

     segment_sizes = [10000, 50000, 100000, 250000] # vectors per segment
     results = []

     for seg_size in segment_sizes:
         cmd = f"./build/benchmarks/bench_compaction_io --seg_size={seg_size} --json"
         out = subprocess.check_output(cmd, shell=True)
         res = json.loads(out)
         results.append({
             "seg_size": seg_size,
             "waf": res["write_amplification_factor"],
             "p99_stall_ms": res["p99_flush_stall_ms"],
             "recall_at_10": res["recall_at_10"]
         })

     # Generate Figure 2: Write Amplification vs Recall Preservation
     fig, ax1 = plt.subplots(figsize=(8, 5))
     ax1.plot([r["seg_size"] for r in results], [r["waf"] for r in results], 'b-o', label="WAF")
     ax1.set_xlabel("Segment Size (vectors)")
     ax1.set_ylabel("Write Amplification Factor", color='b')

     ax2 = ax1.twinx()
     ax2.plot([r["seg_size"] for r in results], [r["recall_at_10"] for r in results], 'g-s', label="Recall@10")
     ax2.set_ylabel("Search Recall@10", color='g')

     plt.title("LSM Compaction Trade-offs: I/O Amplification vs Search Recall")
     plt.grid(True)
     plt.savefig("research/2026-11-late-interaction-lsm/figures/fig2_compaction_waf_vs_recall.png", dpi=300)
     ```
