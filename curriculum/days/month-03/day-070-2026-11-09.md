* **Sunday Research Focus: Milestone 2 (Dynamic Ingestion vs Query Latency Sweep)**:
  1. **Implement Automated Streaming Sweep Harness** in `scripts/sweep_lsm_throughput.py`:
     ```python
     import matplotlib.pyplot as plt
     import numpy as np
     import subprocess
     import json

     write_rates = [100, 500, 1000, 2500, 5000, 10000] # docs/sec
     methods = ["plaid_inverted_lists", "muvera_fde_lsm"]
     results = []

     for method in methods:
         for rate in write_rates:
             cmd = f"./build/benchmarks/bench_concurrent_lsm --method={method} --rate={rate} --duration=30 --json"
             out = subprocess.check_output(cmd, shell=True)
             res = json.loads(out)
             results.append({
                 "method": method,
                 "write_rate": rate,
                 "p99_query_latency_ms": res["p99_latency_ms"],
                 "actual_write_throughput": res["actual_write_tps"]
             })

     # Generate Figure 1: Concurrent Write Rate vs p99 Search Latency
     plt.figure(figsize=(8, 5))
     for method in methods:
         m_res = [r for r in results if r["method"] == method]
         plt.plot([r["write_rate"] for r in m_res], [r["p99_query_latency_ms"] for r in m_res],
                  marker='o', label="PLAID Inverted Lists" if "plaid" in method else "MUVERA LSM")
     plt.xlabel("Ingestion Rate (docs/sec)")
     plt.ylabel("p99 Query Latency (ms)")
     plt.title("Search Latency Under Concurrent Streaming Ingestion")
     plt.legend()
     plt.grid(True)
     plt.savefig("research/2026-11-late-interaction-lsm/figures/fig1_streaming_write_vs_search_latency.png", dpi=300)
     ```
