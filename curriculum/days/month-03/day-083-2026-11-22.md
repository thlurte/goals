## 🧠 14:00 – 18:00 | Deep Learning from Scratch (Python / uv)
* **Workspace**: `~/personal/dl-track/onnx-export-pipeline`
* **Implementation Tasks**:
  1. **Bi-Encoder ONNX Export & Verification** in `src/export_biencoder_onnx.py`:
     ```python
     import torch
     import torch.nn as nn
     import torch.nn.functional as F
     import onnxruntime as ort
     import numpy as np

     class SimpleBiEncoder(nn.Module):
         def __init__(self, vocab_size: int = 1000, dim: int = 128):
             super().__init__()
             self.emb = nn.Embedding(vocab_size, dim)
             self.linear = nn.Linear(dim, dim)

         def forward(self, input_ids: torch.Tensor) -> torch.Tensor:
             x = self.emb(input_ids) # [B, Seq, D]
             pooled = torch.mean(x, dim=1) # [B, D]
             out = self.linear(pooled)
             return F.normalize(out, p=2, dim=-1)

     # 1. Export PyTorch Model to ONNX
     model = SimpleBiEncoder()
     model.eval()
     dummy_input = torch.randint(0, 1000, (4, 32), dtype=torch.long)

     torch.onnx.export(
         model,
         dummy_input,
         "biencoder.onnx",
         input_names=["input_ids"],
         output_names=["embeddings"],
         dynamic_axes={"input_ids": {0: "batch_size", 1: "seq_len"}, "embeddings": {0: "batch_size"}},
         opset_version=17
     )

     # 2. Verify Numerical Parity with ONNX Runtime
     session = ort.InferenceSession("biencoder.onnx")
     ort_inputs = {"input_ids": dummy_input.numpy()}
     ort_out = session.run(None, ort_inputs)[0]

     with torch.no_grad():
         torch_out = model(dummy_input).numpy()

     max_diff = np.max(np.abs(torch_out - ort_out))
     cos_sim = np.dot(torch_out[0], ort_out[0]) / (np.linalg.norm(torch_out[0]) * np.linalg.norm(ort_out[0]))
     print(f"Max Absolute Difference: {max_diff:.8f}")
     print(f"Cosine Similarity Parity: {cos_sim:.8f}")
     assert max_diff < 1e-5
     assert cos_sim > 0.99999
     ```
