     ```python
     import subprocess
     import json
     import matplotlib.pyplot as plt

     # Run end-to-end Pareto sweep comparing GPU IVF, DiskANN io_uring, and In-RAM HNSW
     cmd = "python scripts/run_full_pareto_evaluation.py --dataset=deep10m --json"
     out = subprocess.check_output(cmd, shell=True)
     data = json.loads(out)

     # Plot Figure 3: QPS vs Recall@10 Pareto Curves
     plt.figure(figsize=(9, 6))
     plt.plot(data["gpu_ivf"]["recall"], data["gpu_ivf"]["qps"], marker="o", label="GPU IVF-PQ (In-VRAM)")
     plt.plot(data["diskann"]["recall"], data["diskann"]["qps"], marker="s", label="DiskANN (io_uring NVMe SSD)")
     plt.plot(data["hnsw"]["recall"], data["hnsw"]["qps"], marker="^", label="HNSW (In-DRAM)")
     plt.xlabel("Recall@10")
     plt.ylabel("Query Throughput (QPS)")
     plt.title("Billion-Scale Retrieval Frontiers: In-VRAM GPU vs NVMe DiskANN (DEEP10M)")
     plt.legend()
     plt.grid(True)
     plt.savefig("research/2026-12-three-paths-spine/figures/fig3_qps_vs_recall_pareto.png", dpi=300)
     ```
