  1. **Hybrid Dense-Sparse RRF Search Pipeline** in `src/hybrid_search.py`:
     ```python
     import torch
     import numpy as np
     from typing import List, Dict, Tuple

     def reciprocal_rank_fusion(
         dense_ranks: List[int],
         sparse_ranks: List[int],
         k_rrf: float = 60.0
     ) -> List[Tuple[int, float]]:
         """
         Reciprocal Rank Fusion (RRF) combining dense and sparse ranking lists.
         """
         scores: Dict[int, float] = {}

         for rank, doc_id in enumerate(dense_ranks):
             scores[doc_id] = scores.get(doc_id, 0.0) + 1.0 / (k_rrf + (rank + 1))

         for rank, doc_id in enumerate(sparse_ranks):
             scores[doc_id] = scores.get(doc_id, 0.0) + 1.0 / (k_rrf + (rank + 1))

         sorted_results = sorted(scores.items(), key=lambda x: x[1], reverse=True)
         return sorted_results
     ```
