## 📖 05:30 – 06:30 | Systems & Architecture Reading (60 min)
* **Book / Paper**: *FlashAttention: Fast and Memory-Efficient Exact Attention with IO-Awareness* §3 (*Online Softmax Algorithm & Register Rescaling*)
* **Core Microarchitectural Concepts**:
  1. **The FlashAttention Online Softmax Dynamic Rescaling Math**:
     * Let current running block statistics be row maximum $m^{(j)}$ and normalizer $\ell^{(j)}$.
     * When processing incoming Key/Value block $j+1$ with local maximum $\tilde{m}$:
       $$m^{(j+1)} = \max(m^{(j)}, \tilde{m})$$
       $$\ell^{(j+1)} = e^{m^{(j)} - m^{(j+1)}} \ell^{(j)} + e^{\tilde{m} - m^{(j+1)}} \tilde{\ell}$$
       $$O^{(j+1)} = \operatorname{diag}\left(e^{m^{(j)} - m^{(j+1)}}\right) O^{(j)} + e^{\tilde{m} - m^{(j+1)}} \tilde{P} V_{j+1}$$
  2. **Register-Resident Accumulators**:
     * $O^{(j)}$, $m^{(j)}$, and $\ell^{(j)}$ reside in registers throughout the entire sequence loop, completely avoiding intermediate DRAM writes.

     import torch
     import time
     import json
     import matplotlib.pyplot as plt
     from src.flash_attn_pure import flash_attention_v1_reference

     seq_lens = [512, 1024, 2048, 4096, 8192]
     head_dim = 64
     batch_size = 4
     num_heads = 8

     results = []

     for n in seq_lens:
         Q = torch.randn(batch_size, num_heads, n, head_dim, device="cuda")
         K = torch.randn(batch_size, num_heads, n, head_dim, device="cuda")
         V = torch.randn(batch_size, num_heads, n, head_dim, device="cuda")

         # Warmup
         for _ in range(5):
             _ = torch.nn.functional.scaled_dot_product_attention(Q, K, V)
         torch.cuda.synchronize()

         # Measure PyTorch Native SDPA
         t0 = time.perf_counter()
         for _ in range(20):
             _ = torch.nn.functional.scaled_dot_product_attention(Q, K, V)
         torch.cuda.synchronize()
         sdpa_time_ms = (time.perf_counter() - t0) / 20.0 * 1000.0

         # Measure FlashAttention-1 Custom Kernel
         t0 = time.perf_counter()
         for _ in range(20):
             _ = flash_attention_v1_reference(Q, K, V)
         torch.cuda.synchronize()
         fa_time_ms = (time.perf_counter() - t0) / 20.0 * 1000.0

         results.append({
             "seq_len": n,
             "sdpa_ms": sdpa_time_ms,
             "flash_attn_ms": fa_time_ms,
             "speedup": sdpa_time_ms / fa_time_ms
         })

     with open("research/2026-12-three-paths-spine/results/attention_scaling.json", "w") as f:
         json.dump(results, f, indent=2)

     # Plot Fig 2: Latency vs Sequence Length
     plt.figure(figsize=(8, 5))
     plt.plot(seq_lens, [r["sdpa_ms"] for r in results], marker="o", label="Standard SDPA (PyTorch)")
     plt.plot(seq_lens, [r["flash_attn_ms"] for r in results], marker="s", label="FlashAttention-1 (Ours)")
     plt.xlabel("Sequence Length (N)")
     plt.ylabel("Forward Latency (ms)")
     plt.title("Attention Forward Execution Time vs Sequence Length")
     plt.legend()
     plt.grid(True)
     plt.savefig("research/2026-12-three-paths-spine/figures/fig2_flashattn_latency_memory.png", dpi=300)

  1. **Implement FlashAttention Online Softmax Register Rescaling Engine** in `include/secan/gpu/online_softmax_update.cuh`:
     ```cuda
     #pragma once
     #include <cuda_runtime.h>
     #include <cmath>

     namespace secan {
     template <int HEAD_DIM>
     struct FlashAttentionAccumulator {
         float m_prev{-1e30f};
         float l_prev{0.0f};
         float o_accum[HEAD_DIM]{0.0f};

         __device__ void update(float m_curr, float l_curr, const float* v_curr_weighted) {
             float m_new = fmaxf(m_prev, m_curr);
             float alpha = expf(m_prev - m_new);
             float beta = expf(m_curr - m_new);

             float l_new = alpha * l_prev + beta * l_curr;

             #pragma unroll
             for (int d = 0; d < HEAD_DIM; ++d) {
                 o_accum[d] = alpha * o_accum[d] + beta * v_curr_weighted[d];
             }

             m_prev = m_new;
             l_prev = l_new;
         }

         __device__ void finalize(float* o_out) const {
             float inv_l = 1.0f / (l_prev + 1e-12f);
             #pragma unroll
             for (int d = 0; d < HEAD_DIM; ++d) {
                 o_out[d] = o_accum[d] * inv_l;
             }
         }
     };
     }
     ```
