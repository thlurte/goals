* **Implementation Tasks**:
  1. **Online Softmax CUDA Kernel** in `src/online_softmax.cu` & `src/binding.cpp`:
     ```cuda
     #include <torch/extension.h>
     #include <cuda_runtime.h>
     #include <cfloat>

     __device__ inline float warp_reduce_max(float val) {
         #pragma unroll
         for (int offset = 16; offset > 0; offset /= 2) {
             val = fmaxf(val, __shfl_down_sync(0xffffffff, val, offset));
         }
         return val;
     }

     __device__ inline float warp_reduce_sum(float val) {
         #pragma unroll
         for (int offset = 16; offset > 0; offset /= 2) {
             val += __shfl_down_sync(0xffffffff, val, offset);
         }
         return val;
     }

     // 1 Warp computes numerically stable Online Softmax for 1 row
     __global__ void online_softmax_kernel(const float* __restrict__ input, float* __restrict__ output,
                                           int batch_size, int vocab_size) {
         int row = blockIdx.x * (blockDim.x / 32) + (threadIdx.x / 32);
         int lane = threadIdx.x % 32;
         if (row >= batch_size) return;

         const float* in_row = input + row * vocab_size;
         float* out_row = output + row * vocab_size;

         // Pass 1: Find Row Max
         float thread_max = -FLT_MAX;
         for (int i = lane; i < vocab_size; i += 32) {
             thread_max = fmaxf(thread_max, in_row[i]);
         }
         float row_max = warp_reduce_max(thread_max);
         row_max = __shfl_sync(0xffffffff, row_max, 0);

         // Pass 2: Compute Exp Sum
         float thread_sum = 0.0f;
         for (int i = lane; i < vocab_size; i += 32) {
             thread_sum += expf(in_row[i] - row_max);
         }
         float row_sum = warp_reduce_sum(thread_sum);
         row_sum = __shfl_sync(0xffffffff, row_sum, 0);

         // Pass 3: Normalize
         float inv_sum = 1.0f / (row_sum + 1e-12f);
         for (int i = lane; i < vocab_size; i += 32) {
             out_row[i] = expf(in_row[i] - row_max) * inv_sum;
         }
     }

     torch::Tensor online_softmax_forward(torch::Tensor input) {
         int B = input.size(0);
         int V = input.size(1);
         auto out = torch::empty_like(input);

         int warps_per_block = 8;
         dim3 block(warps_per_block * 32);
         dim3 grid((B + warps_per_block - 1) / warps_per_block);
         online_softmax_kernel<<<grid, block>>>(input.data_ptr<float>(), out.data_ptr<float>(), B, V);
         return out;
     }

     PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
         m.def("forward", &online_softmax_forward, "Online Softmax CUDA forward");
     }
     ```
