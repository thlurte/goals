## 📖 05:30 – 06:30 | Systems & Architecture Reading (60 min)
* **Book**: *Programming Massively Parallel Processors (PMPP)* — David B. Kirk & Wen-mei W. Hwu (Ch 10)
* **Core Microarchitectural Concepts**:
  1. **In-Register Warp Shuffle Primitives (`__shfl_down_sync`)**:
     * Traditional shared-memory reductions require writing partial sums to SRAM, issuing `__syncthreads()`, and suffering potential bank conflicts ($>30\text{ clock cycles}$).
     * The `__shfl_down_sync(mask, val, offset)` instruction exchanges register values directly across ALUs within the same warp in **1 clock cycle** with zero shared memory traffic and zero sync barriers!
  2. **5-Step Logarithmic Warp Tree Reduction**:
     ```cuda
     #pragma unroll
     for (int offset = 16; offset > 0; offset /= 2) {
         val += __shfl_down_sync(0xffffffff, val, offset);
     }
     ```
     * Reduces 32 thread values down to Lane 0 in exactly 5 instructions.

     ```cuda
     #pragma once
     #include <cuda_runtime.h>

     namespace secan {
     __device__ inline float warp_reduce_sum(float val) {
         #pragma unroll
         for (int offset = 16; offset > 0; offset /= 2) {
             val += __shfl_down_sync(0xffffffff, val, offset);
         }
         return val;
     }

     // 1 Warp computes the dot product / L2 distance of a single high-dimensional vector pair
     __global__ void l2_distance_warp_kernel(
         const float* __restrict__ queries, // [num_queries, dim]
         const float* __restrict__ database,// [num_database, dim]
         float* __restrict__ distances,     // [num_queries, num_database]
         size_t num_queries,
         size_t num_database,
         size_t dim
     ) {
         size_t global_warp_id = (blockIdx.x * blockDim.x + threadIdx.x) / 32;
         size_t lane_id = threadIdx.x % 32;

         size_t total_pairs = num_queries * num_database;
         if (global_warp_id >= total_pairs) return;

         size_t q_idx = global_warp_id / num_database;
         size_t db_idx = global_warp_id % num_database;

         const float* q = queries + q_idx * dim;
         const float* d = database + db_idx * dim;

         // Stride across dimensions within the warp
         float thread_sum = 0.0f;
         for (size_t k = lane_id; k < dim; k += 32) {
             float diff = q[k] - d[k];
             thread_sum += diff * diff;
         }

         // Reduce across 32 warp lanes in registers
         float total_dist = warp_reduce_sum(thread_sum);

         if (lane_id == 0) {
             distances[q_idx * num_database + db_idx] = total_dist;
         }
     }
     }
     ```
