* **Core Microarchitectural Concepts**:
  1. **NVIDIA Tensor Core Architecture & WMMA API**:
     * Hardware Matrix Multiply-Accumulate (MMA) units execute $D = A \cdot B + C$ on matrix tiles (e.g. $16 \times 16 \times 16$ FP16) in a single clock cycle across a warp.
     * `nvcuda::wmma` API maps warp registers directly to Tensor Core execution fragments:
       ```cuda
       wmma::fragment<wmma::matrix_a, 16, 16, 16, half, wmma::row_major> a_frag;
       wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::col_major> b_frag;
       wmma::fragment<wmma::accumulator, 16, 16, 16, float> c_frag;
       wmma::load_matrix_sync(a_frag, ...);
       wmma::mma_sync(c_frag, a_frag, b_frag, c_frag);
       ```
  2. **Asynchronous Memory Copy (`cuda::memcpy_async`)**:
     * Hardware copy engines bypass SM ALUs, streaming data directly from global DRAM into shared memory while ALUs compute previous tiles.

  1. **Pure PyTorch Tiled FlashAttention-1 Reference** in `src/flash_attn_pure.py`:
     ```python
     import torch
     import math

     def flash_attention_v1_reference(Q: torch.Tensor, K: torch.Tensor, V: torch.Tensor, block_m: int = 64, block_n: int = 64) -> torch.Tensor:
         """
         Pure PyTorch block-by-block implementation of FlashAttention-1 forward pass.
         Q, K, V: [B, H, N, d]
         """
         B, H, N, d = Q.shape
         scale = 1.0 / math.sqrt(d)
         O = torch.zeros_like(Q)

         for b in range(B):
             for h in range(H):
                 q_mat = Q[b, h] # [N, d]
                 k_mat = K[b, h] # [N, d]
                 v_mat = V[b, h] # [N, d]
                 o_mat = torch.zeros_like(q_mat)

                 num_q_blocks = (N + block_m - 1) // block_m
                 num_kv_blocks = (N + block_n - 1) // block_n

                 for i in range(num_q_blocks):
                     q_start = i * block_m
                     q_end = min(q_start + block_m, N)
                     Q_i = q_mat[q_start:q_end] # [B_M, d]

                     # Initialize block accumulators in registers
                     m_i = torch.full((Q_i.shape[0], 1), -float("inf"), device=Q.device)
                     l_i = torch.zeros((Q_i.shape[0], 1), device=Q.device)
                     o_i = torch.zeros((Q_i.shape[0], d), device=Q.device)

                     for j in range(num_kv_blocks):
                         kv_start = j * block_n
                         kv_end = min(kv_start + block_n, N)
                         K_j = k_mat[kv_start:kv_end] # [B_N, d]
                         V_j = v_mat[kv_start:kv_end] # [B_N, d]

                         # Block GEMM: S_ij = Q_i K_j^T * scale
                         S_ij = torch.matmul(Q_i, K_j.transpose(0, 1)) * scale # [B_M, B_N]

                         # Block Online Softmax Rescaling
                         m_ij, _ = torch.max(S_ij, dim=1, keepdim=True) # [B_M, 1]
                         m_new = torch.maximum(m_i, m_ij)

                         P_ij_unnorm = torch.exp(S_ij - m_new) # [B_M, B_N]
                         l_ij = torch.sum(P_ij_unnorm, dim=1, keepdim=True) # [B_M, 1]

                         alpha = torch.exp(m_i - m_new)
                         l_new = alpha * l_i + l_ij

                         # Accumulate Output
                         o_i = alpha * o_i + torch.matmul(P_ij_unnorm, V_j)

                         m_i = m_new
                         l_i = l_new

                     o_mat[q_start:q_end] = o_i / l_i

                 O[b, h] = o_mat
         return O
     ```

     ```cuda
     #include "secan/gpu/gpu_buffer.cuh"
     #include <cuda_runtime.h>
     #include <cuda/pipeline>

     namespace secan {
     constexpr int TILE_SIZE = 64;

     // Coalesced double-buffered loading from GMEM to SRAM
     template <typename T>
     __device__ inline void load_tile_async(
         T (*s_dest)[TILE_SIZE],
         const T* g_src,
         int row_stride,
         int tile_rows,
         int tile_cols,
         cuda::pipeline<cuda::thread_scope_block>& pipe
     ) {
         int tid = threadIdx.x;
         int total_threads = blockDim.x;

         for (int i = tid; i < tile_rows * tile_cols; i += total_threads) {
             int r = i / tile_cols;
             int c = i % tile_cols;
             cuda::memcpy_async(&s_dest[r][c], &g_src[r * row_stride + c], sizeof(T), pipe);
         }
     }
     }
     ```
