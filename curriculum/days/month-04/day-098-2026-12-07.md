* **Sunday Research Focus: Milestone 2 (GPU Memory Subsystem & Arithmetic Intensity Roofline)**:
  1. **Automated GPU Memory Bandwidth Sweep Script** in `scripts/sweep_gpu_bandwidth.py`:
     ```python
     import subprocess
     import json
     import matplotlib.pyplot as plt
     import numpy as np

     batch_sizes = [1, 4, 16, 64, 256, 1024]
     dimensions = [64, 128, 384, 768, 1536]
     results = []

     for d in dimensions:
         for b in batch_sizes:
             cmd = f"./build/benchmarks/bench_gpu_batched_warps --dim={d} --batch={b} --json"
             out = subprocess.check_output(cmd, shell=True)
             data = json.loads(out)
             results.append({
                 "dim": d,
                 "batch": b,
                 "achieved_bw_gbs": data["achieved_gbps"],
                 "qps": data["qps"],
                 "arithmetic_intensity": (2.0 * d) / (d * 4.0) # FLOPs per byte loaded
             })

     with open("research/2026-12-three-paths-spine/results/gpu_bandwidth.json", "w") as f:
         json.dump(results, f, indent=2)

     # Plot Fig 1: Achieved Bandwidth vs Batch Size
     plt.figure(figsize=(8, 5))
     for d in dimensions:
         subset = [r for r in results if r["dim"] == d]
         plt.plot([r["batch"] for r in subset], [r["achieved_bw_gbs"] for r in subset], marker="o", label=f"Dim {d}")
     plt.axhline(y=936.0, color="r", linestyle="--", label="Peak Theoretical HBM2 (936 GB/s)")
     plt.xscale("log", base=2)
     plt.xlabel("Batch Size (B)")
     plt.ylabel("Achieved Memory Bandwidth (GB/s)")
     plt.title("GPU Memory Bus Saturation vs Query Batch Size")
     plt.legend()
     plt.grid(True)
     plt.savefig("research/2026-12-three-paths-spine/figures/fig1_gpu_bandwidth_roofline.png", dpi=300)
     ```
