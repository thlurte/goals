  1. **Implement Tombstone Manager & Vacuum Rewirer** in `include/secan/hnsw/tombstone_vacuum.h`:
     ```cpp
     #pragma once
     #include <vector>
     #include <atomic>
     #include <cstdint>
     #include <memory>

     namespace secan {
     class TombstoneManager {
     private:
         size_t total_capacity;
         std::vector<std::atomic<uint64_t>> tombstone_words;
         std::atomic<size_t> deleted_count{0};

     public:
         TombstoneManager(size_t capacity) 
             : total_capacity(capacity), tombstone_words((capacity + 63) / 64) {
             for (auto& w : tombstone_words) w.store(0, std::memory_order_relaxed);
         }

         void mark_deleted(uint32_t node_id) {
             size_t word_idx = node_id / 64;
             uint64_t bit_mask = 1ULL << (node_id % 64);
             uint64_t prev = tombstone_words[word_idx].fetch_or(bit_mask, std::memory_order_relaxed);
             if (!(prev & bit_mask)) {
                 deleted_count.fetch_add(1, std::memory_order_relaxed);
             }
         }

         inline bool is_deleted(uint32_t node_id) const {
             size_t word_idx = node_id / 64;
             uint64_t bit_mask = 1ULL << (node_id % 64);
             return (tombstone_words[word_idx].load(std::memory_order_relaxed) & bit_mask) != 0;
         }

         float tombstone_ratio() const {
             return static_cast<float>(deleted_count.load()) / total_capacity;
         }
     };
     }
     ```
