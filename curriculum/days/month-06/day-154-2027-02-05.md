     #pragma once
     #include <immintrin.h>
     #include <vector>
     #include <cstdint>

     namespace secan {
     class ColumnarRangeIndex {
     private:
         size_t num_vectors;
         std::vector<float> values; // Contiguous columnar attributes

     public:
         ColumnarRangeIndex(size_t n) : num_vectors(n), values(n, 0.0f) {}

         void set_attribute(uint32_t id, float val) {
             values[id] = val;
         }

         std::vector<uint8_t> filter_range_avx2(float min_val, float max_val) const {
             std::vector<uint8_t> bitmask((num_vectors + 7) / 8, 0);
             __m256 v_min = _mm256_set1_ps(min_val);
             __m256 v_max = _mm256_set1_ps(max_val);

             for (size_t i = 0; i < num_vectors; i += 8) {
                 __m256 v = _mm256_loadu_ps(&values[i]);
                 __m256 ge = _mm256_cmp_ps(v, v_min, _CMP_GE_OQ);
                 __m256 le = _mm256_cmp_ps(v, v_max, _CMP_LE_OQ);
                 __m256 pass = _mm256_and_ps(ge, le);
                 int mask = _mm256_movemask_ps(pass);
                 bitmask[i / 8] = static_cast<uint8_t>(mask);
             }
             return bitmask;
         }
     };
     }
