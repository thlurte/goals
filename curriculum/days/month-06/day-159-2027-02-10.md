* **Core Microarchitectural Concepts**:
  1. **ARM Dot Product Execution Pipeline**:
     * `vdotq_u32(acc, a, b)` consumes two `uint8x16_t` registers, computing four 4-element dot products and accumulating into `uint32x4_t` in a single clock cycle.
     * Processes 16 quantized bytes ($128\text{ bits}$) per cycle per execution pipe.
  2. **Zero-Overhead SQ8 Asymmetric Distance**:
     * Interleaving `vdotq_u32` with precomputed scalar norms allows full 8-bit integer quantized L2 distance evaluation at memory bus saturation speed ($>40\text{ GB/s}$).

     ```cpp
     #if defined(__ARM_NEON) || defined(__aarch64__)
     #include <arm_neon.h>
     #include <cstddef>

     namespace secan::simd {
     inline int32_t dot_product_sq8_neon(const uint8_t* a, const uint8_t* b, size_t dim) {
         uint32x4_t acc0 = vdupq_n_u32(0);
         uint32x4_t acc1 = vdupq_n_u32(0);

         size_t i = 0;
         for (; i + 31 < dim; i += 32) {
             uint8x16_t va0 = vld1q_u8(a + i);
             uint8x16_t vb0 = vld1q_u8(b + i);
             #if defined(__ARM_FEATURE_DOTPROD)
             acc0 = vdotq_u32(acc0, va0, vb0);
             #else
             uint16x8_t prod_low = vmull_u8(vget_low_u8(va0), vget_low_u8(vb0));
             uint16x8_t prod_high = vmull_u8(vget_high_u8(va0), vget_high_u8(vb0));
             acc0 = vaddw_u16(acc0, vget_low_u16(prod_low));
             #endif

             uint8x16_t va1 = vld1q_u8(a + i + 16);
             uint8x16_t vb1 = vld1q_u8(b + i + 16);
             #if defined(__ARM_FEATURE_DOTPROD)
             acc1 = vdotq_u32(acc1, va1, vb1);
             #endif
         }

         uint32x4_t acc = vaddq_u32(acc0, acc1);
         return static_cast<int32_t>(vaddvq_u32(acc));
     }
     }
     #endif
     ```
