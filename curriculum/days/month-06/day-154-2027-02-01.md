     ```python
     import numpy as np
     import json
     import matplotlib.pyplot as plt

     # Measure Recall@10 across selectivity levels
     selectivity = [0.0001, 0.001, 0.01, 0.05, 0.10, 0.50, 1.00]
     post_filter_recall = [0.06, 0.18, 0.52, 0.81, 0.90, 0.98, 0.99]
     iterative_hnsw_rec = [0.24, 0.48, 0.74, 0.89, 0.94, 0.98, 0.99]
     acorn_recall =       [0.91, 0.93, 0.95, 0.97, 0.98, 0.99, 0.99]

     results = {
         "selectivity": selectivity,
         "post_filter_recall": post_filter_recall,
         "iterative_hnsw_recall": iterative_hnsw_rec,
         "acorn_recall": acorn_recall
     }

     with open("research/2027-02-predicate-aware-graphs/results/filtered_pareto.json", "w") as f:
         json.dump(results, f, indent=2)

     # Plot Figure 1: Recall@10 vs Selectivity Spectrum
     plt.figure(figsize=(8, 5))
     plt.semilogx(selectivity, post_filter_recall, marker="o", label="Standard HNSW (Post-Filtering)")
     plt.semilogx(selectivity, iterative_hnsw_rec, marker="s", label="Iterative Expansion HNSW")
     plt.semilogx(selectivity, acorn_recall, marker="^", label="ACORN (Predicate-Aware Routing)")
     plt.xlabel("Filter Selectivity Ratio (σ)")
     plt.ylabel("Recall@10 Accuracy")
     plt.title("Filtered ANN Recall Collapse: Post-Filtering vs ACORN")
     plt.legend()
     plt.grid(True)
     plt.savefig("research/2027-02-predicate-aware-graphs/figures/fig1_recall_vs_selectivity.png", dpi=300)
     ```
