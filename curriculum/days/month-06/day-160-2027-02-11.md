     ```cpp
     #pragma once
     #include <cstdint>
     #if defined(__x86_64__) || defined(_M_X64)
     #include <cpuid.h>
     #elif defined(__aarch64__) && defined(__linux__)
     #include <sys/auxv.h>
     #include <asm/hwcap.h>
     #elif defined(__APPLE__)
     #include <sys/sysctl.h>
     #endif

     namespace secan::simd {
     struct CpuCapabilities {
         bool has_avx2{false};
         bool has_avx512f{false};
         bool has_neon{false};
         bool has_neon_dotprod{false};
     };

     inline CpuCapabilities detect_cpu_features() {
         CpuCapabilities caps;
     #if defined(__x86_64__) || defined(_M_X64)
         unsigned int eax, ebx, ecx, edx;
         if (__get_cpuid_count(7, 0, &eax, &ebx, &ecx, &edx)) {
             caps.has_avx2 = (ebx & (1 << 5)) != 0;
             caps.has_avx512f = (ebx & (1 << 16)) != 0;
         }
     #elif defined(__aarch64__)
         caps.has_neon = true;
         #if defined(__linux__)
         unsigned long hwcaps = getauxval(AT_HWCAP);
         #if defined(HWCAP_ASIMDDP)
         caps.has_neon_dotprod = (hwcaps & HWCAP_ASIMDDP) != 0;
         #endif
         #elif defined(__APPLE__)
         int val = 0; size_t len = sizeof(val);
         if (sysctlbyname("hw.optional.AdvSIMD_DotProd", &val, &len, NULL, 0) == 0) {
             caps.has_neon_dotprod = (val == 1);
         }
         #endif
     #endif
         return caps;
     }
     }
     ```
