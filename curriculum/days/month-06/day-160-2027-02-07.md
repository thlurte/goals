* **Implementation Tasks**:
  1. **Spectral Graph Partitioning & Fiedler Sweep Cut in Pure PyTorch** in `src/spectral_partition.py`:
     ```python
     import torch
     from typing import Tuple

     class SpectralGraphPartitioner:
         def __init__(self, adjacency_matrix: torch.Tensor):
             self.A = adjacency_matrix.float()
             self.n = self.A.shape[0]
             self.degrees = torch.sum(self.A, dim=1)
             self.D_inv_sqrt = torch.diag(1.0 / torch.sqrt(torch.clamp(self.degrees, min=1e-8)))
             # Normalized Laplacian: L_norm = I - D^{-1/2} A D^{-1/2}
             self.L_norm = torch.eye(self.n) - self.D_inv_sqrt @ self.A @ self.D_inv_sqrt

         def compute_fiedler_vector(self) -> torch.Tensor:
             """
             Extract the second smallest eigenvector (Fiedler vector) of L_norm.
             """
             eigenvalues, eigenvectors = torch.linalg.eigh(self.L_norm)
             # Index 1 corresponds to lambda_2
             fiedler_v2 = eigenvectors[:, 1]
             # Un-normalize: y = D^{-1/2} v_2
             return self.D_inv_sqrt @ fiedler_v2

         def fiedler_sweep_cut(self) -> Tuple[torch.Tensor, float]:
             """
             Execute sweep cut over sorted Fiedler coordinates to find minimum conductance cut.
             """
             y = self.compute_fiedler_vector()
             sorted_indices = torch.argsort(y)
             total_vol = torch.sum(self.degrees).item()

             best_conductance = float("inf")
             best_cut_mask = None

             for k in range(1, self.n):
                 S_indices = sorted_indices[:k]
                 mask_S = torch.zeros(self.n, dtype=torch.bool)
                 mask_S[S_indices] = True

                 vol_S = torch.sum(self.degrees[mask_S]).item()
                 vol_S_c = total_vol - vol_S
                 denom = min(vol_S, vol_S_c)
                 if denom == 0:
                     continue

                 # Count cut edges E(S, S^c)
                 cut_edges = torch.sum(self.A[mask_S][:, ~mask_S]).item()
                 conductance = cut_edges / denom

                 if conductance < best_conductance:
                     best_conductance = conductance
                     best_cut_mask = mask_S

             return best_cut_mask, best_conductance
     ```
