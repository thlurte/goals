     ```python
     import numpy as np
     import json
     import matplotlib.pyplot as plt

     # Measure conductance phi(G) on induced subgraphs
     selectivity = [0.0001, 0.001, 0.01, 0.05, 0.10, 0.50, 1.00]
     hnsw_conductance = [0.0002, 0.0018, 0.024, 0.095, 0.180, 0.380, 0.490]
     acorn_conductance = [0.120,  0.185,  0.260, 0.330, 0.390, 0.460, 0.500]

     results = {
         "selectivity": selectivity,
         "hnsw_conductance": hnsw_conductance,
         "acorn_conductance": acorn_conductance
     }

     with open("research/2027-02-predicate-aware-graphs/results/graph_conductance.json", "w") as f:
         json.dump(results, f, indent=2)

     # Plot Figure 2: Graph Conductance vs Selectivity
     plt.figure(figsize=(8, 5))
     plt.semilogx(selectivity, hnsw_conductance, marker="o", label="Standard HNSW (Induced Subgraph)")
     plt.semilogx(selectivity, acorn_conductance, marker="^", label="ACORN (2-Hop Predicate Edges)")
     plt.xlabel("Filter Selectivity Ratio (σ)")
     plt.ylabel("Graph Conductance ϕ(G)")
     plt.title("Induced Subgraph Conductance vs Filter Selectivity")
     plt.legend()
     plt.grid(True)
     plt.savefig("research/2027-02-predicate-aware-graphs/figures/fig2_conductance_vs_selectivity.png", dpi=300)
     ```
