     ```python
     import torch
     import torch.nn as nn
     import torch.nn.functional as F
     from typing import List, Tuple

     class DualEncoder(nn.Module):
         def __init__(self, vocab_size: int = 10000, embed_dim: int = 128):
             super().__init__()
             self.token_embed = nn.Embedding(vocab_size, embed_dim)
             self.proj = nn.Linear(embed_dim, embed_dim, bias=False)

         def forward(self, token_ids: torch.Tensor) -> torch.Tensor:
             # Mean-pooled token embeddings normalized to unit sphere
             embeds = self.token_embed(token_ids)
             pooled = torch.mean(embeds, dim=1)
             return F.normalize(self.proj(pooled), p=2, dim=-1)

     def compute_infonce_loss(q_embeds: torch.Tensor, k_pos_embeds: torch.Tensor, tau: float = 0.07) -> torch.Tensor:
         # [B, B] similarity logits
         logits = torch.matmul(q_embeds, k_pos_embeds.T) / tau
         labels = torch.arange(q_embeds.shape[0], device=q_embeds.device)
         return F.cross_entropy(logits, labels)
     ```
