## 📖 05:30 – 06:30 | Systems & Architecture Reading (60 min)
* **Book**: *Intel 64 and IA-32 Architectures Optimization Reference Manual* & *Intrinsics Guide*
* **Core Microarchitectural Concepts**:
  1. **Integer SIMD Throughput (32 Integers per Instruction)**:
     * A 256-bit YMM register holds **32 packed 8-bit unsigned integers (`uint8_t`)**.
     * Single-instruction integer difference: `_mm256_subs_epu8` (saturated subtraction) and `_mm256_abs_epi8`.
  2. **Multiplication & Widening Accumulation Pipeline**:
     * `_mm256_maddubs_epi16`: Multiplies pairs of 8-bit integers and horizontally adds adjacent pairs to produce 16-bit integers (`int16_t`).
     * `_mm256_madd_epi16`: Multiplies pairs of 16-bit integers and adds adjacent pairs to produce 32-bit integers (`int32_t`), completely preventing arithmetic overflow.
  3. **AVX-VNNI (Vector Neural Network Instructions)**:
     * `_mm256_dpbusd_epi32`: Fuses 8-bit multiply, 16-bit add, and 32-bit accumulate into a single 3-cycle instruction, executing 32 INT8 operations per clock.

     ```python
     import torch
     import torch.nn as nn

     class PatchEmbedding(nn.Module):
         def __init__(self, img_size: int = 224, patch_size: int = 16, in_channels: int = 3, d_model: int = 768):
             super().__init__()
             self.img_size = img_size
             self.patch_size = patch_size
             self.num_patches = (img_size // patch_size) ** 2

             # Patch projection using convolution
             self.proj = nn.Conv2d(in_channels, d_model, kernel_size=patch_size, stride=patch_size)

         def forward(self, x: torch.Tensor) -> torch.Tensor:
             # Input: [B, C, H, W] -> Output: [B, num_patches, d_model]
             x = self.proj(x)  # [B, d_model, H/P, W/P]
             x = x.flatten(2).transpose(1, 2)  # [B, num_patches, d_model]
             return x

     class VisionTransformer(nn.Module):
         def __init__(self, img_size: int = 224, patch_size: int = 16, in_channels: int = 3,
                      d_model: int = 768, num_layers: int = 12, num_heads: int = 12,
                      d_ff: int = 3072, num_classes: int = 1000):
             super().__init__()
             self.patch_embed = PatchEmbedding(img_size, patch_size, in_channels, d_model)
             self.cls_token = nn.Parameter(torch.zeros(1, 1, d_model))
             self.pos_embed = nn.Parameter(torch.zeros(1, self.patch_embed.num_patches + 1, d_model))

             encoder_layer = nn.TransformerEncoderLayer(
                 d_model=d_model, nhead=num_heads, dim_feedforward=d_ff,
                 activation="gelu", batch_first=True, norm_first=True
             )
             self.encoder = nn.TransformerEncoder(encoder_layer, num_layers=num_layers)
             self.norm = nn.LayerNorm(d_model)
             self.head = nn.Linear(d_model, num_classes)

             nn.init.trunc_normal_(self.pos_embed, std=0.02)
             nn.init.trunc_normal_(self.cls_token, std=0.02)

         def forward(self, x: torch.Tensor) -> torch.Tensor:
             B = x.shape[0]
             x = self.patch_embed(x)
             # Prepend [CLS] token
             cls_tokens = self.cls_token.expand(B, -1, -1)
             x = torch.cat((cls_tokens, x), dim=1)
             x = x + self.pos_embed
             x = self.encoder(x)
             x = self.norm(x[:, 0])  # Extract [CLS] representation
             return self.head(x)
     ```

     ```cpp
     #include <immintrin.h>
     #include <cstdint>
     #include <cstddef>

     namespace secan {
     inline uint32_t l2_squared_sq8_avx2(const uint8_t* a, const uint8_t* b, size_t dim) {
         __m256i sum32 = _mm256_setzero_si256();
         size_t i = 0;

         // Process 32 uint8 elements per iteration
         for (; i + 32 <= dim; i += 32) {
             __m256i va = _mm256_loadu_si256(reinterpret_cast<const __m256i*>(a + i));
             __m256i vb = _mm256_loadu_si256(reinterpret_cast<const __m256i*>(b + i));

             // diff = |va - vb|
             __m256i max_v = _mm256_max_epu8(va, vb);
             __m256i min_v = _mm256_min_epu8(va, vb);
             __m256i diff  = _mm256_sub_epi8(max_v, min_v);

             // Unpack and square: diff_lo, diff_hi
             __m256i zero = _mm256_setzero_si256();
             __m256i d_lo = _mm256_unpacklo_epi8(diff, zero);
             __m256i d_hi = _mm256_unpackhi_epi8(diff, zero);

             __m256i sq_lo = _mm256_madd_epi16(d_lo, d_lo);
             __m256i sq_hi = _mm256_madd_epi16(d_hi, d_hi);

             sum32 = _mm256_add_epi32(sum32, _mm256_add_epi32(sq_lo, sq_hi));
         }

         // Horizontal reduction of 8x int32 in sum32
         __m128i lo128 = _mm256_castsi256_si128(sum32);
         __m128i hi128 = _mm256_extracti128_si256(sum32, 1);
         __m128i sum128 = _mm_add_epi32(lo128, hi128);
         sum128 = _mm_hadd_epi32(sum128, sum128);
         sum128 = _mm_hadd_epi32(sum128, sum128);
         uint32_t total = static_cast<uint32_t>(_mm_cvtsi128_si32(sum128));

         // Scalar tail
         for (; i < dim; ++i) {
             int32_t d = static_cast<int32_t>(a[i]) - static_cast<int32_t>(b[i]);
             total += static_cast<uint32_t>(d * d);
         }
         return total;
     }
     }
     ```
