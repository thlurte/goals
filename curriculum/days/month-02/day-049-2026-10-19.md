* **Sunday Research Focus: Milestone 3 (Anisotropic Loss Ablation & Hubness Skewness)**:
  1. **Implement Hubness & Anisotropic Loss Sweep** in `scripts/ablate_anisotropic_loss.py`:
     ```python
     import numpy as np
     import matplotlib.pyplot as plt
     import subprocess
     import json
     from scipy.stats import skew

     h_weights = [1.0, 2.0, 5.0, 10.0, 20.0]
     models = ["openai_v3_small", "clip_vit_b32", "bge_large"]
     results = []

     for model in models:
         for h in h_weights:
             cmd = f"./build/benchmarks/bench_aniso --model={model} --h={h} --json"
             out = subprocess.check_output(cmd, shell=True)
             res = json.loads(out)
             # res["k_occurrence"]: Array of size N recording how many times vector i appears in top-10
             s_nk = float(skew(res["k_occurrence"]))
             results.append({
                 "model": model,
                 "h": h,
                 "recall_at_10": res["recall_at_10"],
                 "hubness_skewness": s_nk,
             })

     # Generate Figure 2: Hubness Skewness vs Anisotropic Weight h
     plt.figure(figsize=(8, 5))
     for model in models:
         m_res = [r for r in results if r["model"] == model]
         plt.plot([r["h"] for r in m_res], [r["hubness_skewness"] for r in m_res], marker='s', label=model)
     plt.xlabel("Anisotropic Loss Weight (h)")
     plt.ylabel("Hubness Skewness ($S_{N_k}$)")
     plt.title("Mitigation of Quantization Hubness via Anisotropic Loss")
     plt.legend()
     plt.grid(True)
     plt.savefig("research/2026-10-anisotropy-hubness-bits/figures/fig2_hubness_vs_anisotropic_h.png", dpi=300)
     ```
