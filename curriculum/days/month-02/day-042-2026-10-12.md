     ```python
     import numpy as np
     import matplotlib.pyplot as plt
     import subprocess
     import json

     # Compare isotropic (SIFT1M) vs anisotropic (CLIP 768-D) datasets
     datasets = ["sift1m", "clip_vit_b32"]
     subspaces = [4, 8, 16, 32]
     results = []

     for ds in datasets:
         for m in subspaces:
             cmd = f"./build/benchmarks/bench_pq --dataset={ds} --M={m} --json"
             out = subprocess.check_output(cmd, shell=True)
             res = json.loads(out)
             results.append({
                 "dataset": ds,
                 "M": m,
                 "mse": res["reconstruction_mse"],
                 "recall_at_10": res["recall_at_10"],
                 "qps": res["qps"]
             })

     # Generate Figure 1: Quantization MSE vs Subspace Count
     plt.figure(figsize=(8, 5))
     for ds in datasets:
         ds_res = [r for r in results if r["dataset"] == ds]
         plt.plot([r["M"] for r in ds_res], [r["mse"] for r in ds_res], marker='o', label=ds)
     plt.xlabel("Subspace Count (M)")
     plt.ylabel("Reconstruction MSE")
     plt.title("Quantization Error: Isotropic (SIFT) vs Anisotropic (CLIP)")
     plt.legend()
     plt.grid(True)
     plt.savefig("research/2026-10-anisotropy-hubness-bits/figures/fig1_anisotropy_vs_mse.png", dpi=300)
     ```
