     ```python
     import numpy as np
     import json

     def analyze_embedding_geometry(matrix: np.ndarray, name: str) -> dict:
         # Normalize vectors
         norms = np.linalg.norm(matrix, axis=1, keepdims=True)
         normed = matrix / np.maximum(norms, 1e-12)

         # Sample 10,000 pairs to compute mean cosine angle
         idx1 = np.random.choice(len(normed), 10000)
         idx2 = np.random.choice(len(normed), 10000)
         cosines = np.sum(normed[idx1] * normed[idx2], axis=1)

         # Singular Value Decomposition to measure variance decay
         _, s, _ = np.linalg.svd(normed - np.mean(normed, axis=0), full_matrices=False)
         var_explained_top10 = np.sum(s[:10]**2) / np.sum(s**2)

         return {
             "dataset": name,
             "mean_pairwise_cosine": float(np.mean(cosines)),
             "std_pairwise_cosine": float(np.std(cosines)),
             "top10_singular_variance_ratio": float(var_explained_top10),
         }
     ```
