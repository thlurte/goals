  1. **InfoNCE Contrastive Loss with In-Batch Negatives** in `src/loss.py`:
     ```python
     import torch
     import torch.nn as nn
     import torch.nn.functional as F

     class InfoNCELoss(nn.Module):
         def __init__(self, temperature: float = 0.05):
             super().__init__()
             self.temperature = temperature

         def forward(self, query_embs: torch.Tensor, doc_embs: torch.Tensor) -> torch.Tensor:
             # Normalize embeddings to unit sphere
             q_norm = F.normalize(query_embs, p=2, dim=1)
             d_norm = F.normalize(doc_embs, p=2, dim=1)

             # Similarity matrix: [B, B]
             sim_matrix = torch.matmul(q_norm, d_norm.T) / self.temperature

             # Diagonal elements are the positive pairs
             labels = torch.arange(query_embs.shape[0], device=query_embs.device)
             loss = F.cross_entropy(sim_matrix, labels)
             return loss
     ```
  2. **Decoupled Weight Decay AdamW Optimizer from Scratch** in `src/optimizer.py`:
     ```python
     import math
     import torch

     class AdamW:
         def __init__(self, params, lr: float = 1e-3, betas: tuple[float, float] = (0.9, 0.999),
                      eps: float = 1e-8, weight_decay: float = 0.01):
             self.params = list(params)
             self.lr = lr
             self.beta1, self.beta2 = betas
             self.eps = eps
             self.weight_decay = weight_decay
             self.step_num = 0
             self.exp_avg = [torch.zeros_like(p.data) for p in self.params]
             self.exp_avg_sq = [torch.zeros_like(p.data) for p in self.params]

         def step(self):
             self.step_num += 1
             with torch.no_grad():
                 for p, m, v in zip(self.params, self.exp_avg, self.exp_avg_sq):
                     if p.grad is None:
                         continue
                     grad = p.grad

                     # 1. Decoupled Weight Decay
                     p.mul_(1.0 - self.lr * self.weight_decay)

                     # 2. Update biased 1st and 2nd moment estimates
                     m.mul_(self.beta1).add_(grad, alpha=1.0 - self.beta1)
                     v.mul_(self.beta2).addcmul_(grad, grad, value=1.0 - self.beta2)

                     # 3. Bias correction
                     bias_correction1 = 1.0 - self.beta1 ** self.step_num
                     bias_correction2 = 1.0 - self.beta2 ** self.step_num

                     step_size = self.lr / bias_correction1
                     denom = (v.sqrt() / math.sqrt(bias_correction2)).add_(self.eps)

                     # 4. Parameter update
                     p.addcdiv_(m, denom, value=-step_size)

         def zero_grad(self):
             for p in self.params:
                 if p.grad is not None:
                     p.grad.detach_()
                     p.grad.zero_()
     ```
