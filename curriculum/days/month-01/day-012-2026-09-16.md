* **Core Microarchitectural Concepts**:
  1. **FMA Port Mapping on Modern x86 Silicon**:
     * Intel Skylake / IceLake / TigerLake: 2 FMA execution units located on **Port 0** and **Port 1**.
     * AMD Zen 3 / Zen 4: 2 FMA execution units on **Pipe 0** and **Pipe 1**.
  2. **Peak Theoretical Single-Core Compute Throughput**:
     $$\text{Peak GFLOPS} = 2 \text{ FMA units} \times 8 \text{ floats (256-bit)} \times 2 \text{ (mul+add)} \times \text{Clock Frequency (GHz)}$$
     * On a 4.0 GHz CPU, a single core can execute $128 \text{ GFLOPS}$ of single-precision floating point arithmetic if L1 cache loads do not bottleneck the pipeline.
  3. **Fused 1-Pass vs 2-Pass Memory Access**:
     * Computing dot product, norm $A$, and norm $B$ in a single loop pass loads vector $a$ and $b$ from memory once, reducing cache bus traffic by $66\%$.

     ```python
         def __init__(self, d_model: int, num_q_heads: int, num_kv_heads: int, d_ff: int, pre_ln: bool = True):
             super().__init__()
             self.pre_ln = pre_ln
             self.ln1 = nn.LayerNorm(d_model)
             self.attn = GroupedQueryAttention(d_model, num_q_heads, num_kv_heads)
             self.ln2 = nn.LayerNorm(d_model)
             self.mlp = nn.Sequential(
                 nn.Linear(d_model, d_ff),
                 nn.GELU(),
                 nn.Linear(d_ff, d_model)
             )

         def forward(self, x: torch.Tensor) -> torch.Tensor:
             if self.pre_ln:
                 x = x + self.attn(self.ln1(x))
                 x = x + self.mlp(self.ln2(x))
             else:
                 x = self.ln1(x + self.attn(x))
                 x = self.ln2(x + self.mlp(x))
             return x
     ```

  1. **Implement Fused 1-Pass AVX2 Cosine Distance** in `src/distance_avx2.cpp`:
     ```cpp
     #include <immintrin.h>
     #include <cmath>

     namespace secan {
     inline float horizontal_sum_avx2(__m256 v) {
         __m128 lo = _mm256_castps256_ps128(v);
         __m128 hi = _mm256_extractf128_ps(v, 1);
         __m128 sum128 = _mm_add_ps(lo, hi);
         sum128 = _mm_hadd_ps(sum128, sum128);
         sum128 = _mm_hadd_ps(sum128, sum128);
         return _mm_cvtss_f32(sum128);
     }

     inline float cosine_distance_avx2(const float* a, const float* b, size_t dim) {
         __m256 sum_dot = _mm256_setzero_ps();
         __m256 sum_na  = _mm256_setzero_ps();
         __m256 sum_nb  = _mm256_setzero_ps();

         size_t i = 0;
         for (; i + 8 <= dim; i += 8) {
             __m256 va = _mm256_loadu_ps(a + i);
             __m256 vb = _mm256_loadu_ps(b + i);
             sum_dot = _mm256_fmadd_ps(va, vb, sum_dot);
             sum_na  = _mm256_fmadd_ps(va, va, sum_na);
             sum_nb  = _mm256_fmadd_ps(vb, vb, sum_nb);
         }

         float dot  = horizontal_sum_avx2(sum_dot);
         float na   = horizontal_sum_avx2(sum_na);
         float nb   = horizontal_sum_avx2(sum_nb);

         // Scalar tail
         for (; i < dim; ++i) {
             dot += a[i] * b[i];
             na  += a[i] * a[i];
             nb  += b[i] * b[i];
         }

         float denom = std::sqrt(na) * std::sqrt(nb);
         return (denom > 1e-12f) ? (1.0f - dot / denom) : 0.0f;
     }
     }
     ```
