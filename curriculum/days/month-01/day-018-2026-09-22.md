## 📖 05:30 – 06:30 | Systems & Architecture Reading (60 min)
* **Book**: *Computer Systems: A Programmer's Perspective (3rd Edition)* (CS:APP) — R. Bryant & D. O'Hallaron
* **Sections**: §6.4–6.5 (*Cache Memories: Set Associativity, Conflict Misses & Cache Blocking*)
* **Core Microarchitectural Concepts**:
  1. **Cache Structural Parameters $(S, E, B, m)$**:
     * $S = 2^s$ cache sets.
     * $E$ cache lines per set ($E$-way set associative, e.g. 8-way L1, 16-way L3).
     * $B = 2^b$ bytes per cache line ($B = 64\text{ bytes}$).
  2. **The 3 C's of Cache Misses**:
     * **Compulsory (Cold) Miss**: First access to a 64-byte block.
     * **Capacity Miss**: Working set size exceeds total cache capacity.
     * **Conflict Miss**: Two frequently accessed memory regions map to the exact same cache set $s = (\text{address} \gg b) \bmod S$.
  3. **Cache Tiling / Blocking for Batched Retrieval**:
     * When searching $Q$ query vectors simultaneously, scanning one tile of database vectors of size $B_{\text{tile}} \le \text{L2 Capacity}$ ($256\text{ KB}\text{--}1\text{ MB}$) allows the tile to remain resident in L2 cache across all $Q$ queries, replacing $Q$ DRAM passes with $1$ DRAM pass.

     ```python
     import math

     class Value:
         def __init__(self, data: float, _children=(), _op=''):
             self.data = float(data)
             self.grad = 0.0
             self._backward = lambda: None
             self._prev = set(_children)
             self._op = _op

         def __add__(self, other):
             other = other if isinstance(other, Value) else Value(other)
             out = Value(self.data + other.data, (self, other), '+')
             def _backward():
                 self.grad += out.grad
                 other.grad += out.grad
             out._backward = _backward
             return out

         def __mul__(self, other):
             other = other if isinstance(other, Value) else Value(other)
             out = Value(self.data * other.data, (self, other), '*')
             def _backward():
                 self.grad += other.data * out.grad
                 other.grad += self.data * out.grad
             out._backward = _backward
             return out

         def __pow__(self, other: float):
             assert isinstance(other, (int, float))
             out = Value(self.data ** other, (self,), f'**{other}')
             def _backward():
                 self.grad += (other * (self.data ** (other - 1))) * out.grad
             out._backward = _backward
             return out

         def exp(self):
             out = Value(math.exp(self.data), (self,), 'exp')
             def _backward():
                 self.grad += out.data * out.grad
                 out._backward = _backward
             return out

         def tanh(self):
             x = self.data
             t = (math.exp(2*x) - 1) / (math.exp(2*x) + 1)
             out = Value(t, (self,), 'tanh')
             def _backward():
                 self.grad += (1.0 - t**2) * out.grad
             out._backward = _backward
             return out

         def backward(self):
             # Topological sort
             topo = []
             visited = set()
             def build_topo(v):
                 if v not in visited:
                     visited.add(v)
                     for child in v._prev:
                         build_topo(child)
                     topo.append(v)
             build_topo(self)

             self.grad = 1.0
             for v in reversed(topo):
                 v._backward()

         def __repr__(self):
             return f"Value(data={self.data:.4f}, grad={self.grad:.4f})"
     ```

     ```cpp
     #include "secan/distance_avx2.h"
     #include "secan/utils/io.h"
     #include <queue>
     #include <vector>

     namespace secan {
     struct TopKQueryResult {
         std::vector<int32_t> indices;
         std::vector<float> distances;
     };

     // Batched query scan over database with L2 cache tiling
     std::vector<TopKQueryResult> batch_linear_scan_tiled(
         const float* queries, size_t num_queries,
         const float* base, size_t num_base,
         size_t dim, size_t k,
         size_t tile_size = 2048 // 2048 vectors * 128 floats * 4B = 1MB L2/L3 tile
     ) {
         // Priority queues maintaining top-k for each query
         std::vector<std::priority_queue<std::pair<float, int32_t>>> heaps(num_queries);

         // Outer loop over database tiles (streamed once from DRAM into L2)
         for (size_t t = 0; t < num_base; t += tile_size) {
             size_t cur_tile_size = std::min(tile_size, num_base - t);
             const float* tile_data = base + t * dim;

             // Inner loop over queries (reusing the tile resident in cache)
             for (size_t q = 0; q < num_queries; ++q) {
                 const float* q_vec = queries + q * dim;
                 auto& heap = heaps[q];

                 for (size_t i = 0; i < cur_tile_size; ++i) {
                     int32_t global_idx = static_cast<int32_t>(t + i);
                     const float* b_vec = tile_data + i * dim;
                     float dist = l2_squared_avx2_unrolled4(q_vec, b_vec, dim);

                     if (heap.size() < k) {
                         heap.emplace(dist, global_idx);
                     } else if (dist < heap.top().first) {
                         heap.pop();
                         heap.emplace(dist, global_idx);
                     }
                 }
             }
         }

         std::vector<TopKQueryResult> results(num_queries);
         for (size_t q = 0; q < num_queries; ++q) {
             auto& heap = heaps[q];
             results[q].indices.resize(heap.size());
             results[q].distances.resize(heap.size());
             for (int i = static_cast<int>(heap.size()) - 1; i >= 0; --i) {
                 results[q].distances[i] = heap.top().first;
                 results[q].indices[i] = heap.top().second;
                 heap.pop();
             }
         }
         return results;
     }
     }
     ```
