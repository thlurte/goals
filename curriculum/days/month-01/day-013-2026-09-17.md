## 📖 05:30 – 06:30 | Systems & Architecture Reading (60 min)
* **Book**: *Optimizing software in C++* — Agner Fog
* **Sections**: Chapter 13.1–13.3 (*Alignment of Data & Cache-Line Split Penalties*)
* **Core Microarchitectural Concepts**:
  1. **64-Byte Cache Line Boundary Mechanics**:
     * L1/L2/L3 CPU caches operate exclusively on 64-byte aligned chunks.
     * When reading a 32-byte (256-bit YMM) vector that crosses a 64-byte boundary (e.g. from byte 48 to 80), the CPU hardware must issue **two separate cache line requests**, stalling the pipeline by 4–12 clock cycles.
  2. **Aligned vs. Unaligned SIMD Loads**:
     * `_mm256_load_ps`: Assumes the pointer is strictly 32-byte aligned; triggers a General Protection Fault (`#GP`) if misaligned.
     * `_mm256_loadu_ps`: Handles arbitrary alignment, but suffers cache line split penalties when data crosses 64-byte boundaries.
  3. **C++20 Alignment Primitives**:
     * Using `alignas(64)` for stack buffers and `std::aligned_alloc(64, size)` / `posix_memalign` for heap vectors to guarantee cache-line boundary alignment.

     ```cpp
     #include <immintrin.h>
     #include <cstddef>

     namespace secan {
     inline float ip_avx2_unrolled4(const float* a, const float* b, size_t dim) {
         __m256 sum0 = _mm256_setzero_ps();
         __m256 sum1 = _mm256_setzero_ps();
         __m256 sum2 = _mm256_setzero_ps();
         __m256 sum3 = _mm256_setzero_ps();

         size_t i = 0;
         for (; i + 32 <= dim; i += 32) {
             __m256 va0 = _mm256_loadu_ps(a + i);
             __m256 vb0 = _mm256_loadu_ps(b + i);
             sum0 = _mm256_fmadd_ps(va0, vb0, sum0);

             __m256 va1 = _mm256_loadu_ps(a + i + 8);
             __m256 vb1 = _mm256_loadu_ps(b + i + 8);
             sum1 = _mm256_fmadd_ps(va1, vb1, sum1);

             __m256 va2 = _mm256_loadu_ps(a + i + 16);
             __m256 vb2 = _mm256_loadu_ps(b + i + 16);
             sum2 = _mm256_fmadd_ps(va2, vb2, sum2);

             __m256 va3 = _mm256_loadu_ps(a + i + 24);
             __m256 vb3 = _mm256_loadu_ps(b + i + 24);
             sum3 = _mm256_fmadd_ps(va3, vb3, sum3);
         }

         __m256 sum = _mm256_add_ps(_mm256_add_ps(sum0, sum1), _mm256_add_ps(sum2, sum3));

         for (; i + 8 <= dim; i += 8) {
             __m256 va = _mm256_loadu_ps(a + i);
             __m256 vb = _mm256_loadu_ps(b + i);
             sum = _mm256_fmadd_ps(va, vb, sum);
         }

         __m128 lo = _mm256_castps256_ps128(sum);
         __m128 hi = _mm256_extractf128_ps(sum, 1);
         __m128 sum128 = _mm_add_ps(lo, hi);
         sum128 = _mm_hadd_ps(sum128, sum128);
         sum128 = _mm_hadd_ps(sum128, sum128);
         float total = _mm_cvtss_f32(sum128);

         for (; i < dim; ++i) {
             total += a[i] * b[i];
         }
         return total;
     }
     }
     ```
