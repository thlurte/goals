## 📖 05:30 – 06:30 | Systems & Architecture Reading (60 min)
* **Book**: *Computer Systems: A Programmer's Perspective (3rd Edition)* (CS:APP) — R. Bryant & D. O'Hallaron
* **Sections**: §5.8–5.9 (*Loop Unrolling & Multiple Accumulators: Breaking Dependency Chains*)
* **Core Microarchitectural Concepts**:
  1. **Read-After-Write (RAW) Data Dependency Hazard**:
     * In a single-accumulator loop (`sum = _mm256_fmadd_ps(diff, diff, sum)`), the next FMA cannot dispatch until the current FMA finishes its 4-cycle latency.
     * The hardware pipeline sits idle for 3 out of every 4 clock cycles, capping throughput at 0.25 operations per cycle.
  2. **Multi-Register Accumulator Unrolling**:
     * By allocating 4 independent accumulator registers (`sum0`, `sum1`, `sum2`, `sum3`), each subsequent FMA has no data dependency on the preceding instruction.
     * The processor dispatches one FMA every 0.5 cycles across Port 0 and Port 1, achieving full instruction-level parallelism (ILP) and saturating CPU execution ports.

     ```python
     import torch
     import torch.nn as nn
     import math

     class GroupedQueryAttention(nn.Module):
         def __init__(self, d_model: int, num_q_heads: int, num_kv_heads: int):
             super().__init__()
             self.d_model = d_model
             self.num_q_heads = num_q_heads
             self.num_kv_heads = num_kv_heads
             self.head_dim = d_model // num_q_heads
             self.num_queries_per_kv = num_q_heads // num_kv_heads

             self.q_proj = nn.Linear(d_model, d_model, bias=False)
             self.k_proj = nn.Linear(d_model, num_kv_heads * self.head_dim, bias=False)
             self.v_proj = nn.Linear(d_model, num_kv_heads * self.head_dim, bias=False)
             self.out_proj = nn.Linear(d_model, d_model, bias=False)

         def forward(self, x: torch.Tensor, is_causal: bool = True) -> torch.Tensor:
             B, S, D = x.shape
             # Projections
             q = self.q_proj(x).view(B, S, self.num_q_heads, self.head_dim).transpose(1, 2)
             k = self.k_proj(x).view(B, S, self.num_kv_heads, self.head_dim).transpose(1, 2)
             v = self.v_proj(x).view(B, S, self.num_kv_heads, self.head_dim).transpose(1, 2)

             # Repeat KV heads to match Q heads
             k = torch.repeat_interleave(k, self.num_queries_per_kv, dim=1)
             v = torch.repeat_interleave(v, self.num_queries_per_kv, dim=1)

             scores = torch.matmul(q, k.transpose(-2, -1)) / math.sqrt(self.head_dim)
             if is_causal:
                 mask = torch.triu(torch.full((S, S), float("-inf"), device=x.device), diagonal=1)
                 scores = scores + mask

             attn = torch.softmax(scores, dim=-1)
             out = torch.matmul(attn, v).transpose(1, 2).contiguous().view(B, S, D)
             return self.out_proj(out)
     ```

     ```cpp
     #include <immintrin.h>
     #include <cstddef>

     namespace secan {
     inline float l2_squared_avx2_unrolled4(const float* a, const float* b, size_t dim) {
         __m256 sum0 = _mm256_setzero_ps();
         __m256 sum1 = _mm256_setzero_ps();
         __m256 sum2 = _mm256_setzero_ps();
         __m256 sum3 = _mm256_setzero_ps();

         size_t i = 0;
         // Process 32 elements per iteration (4 YMM registers)
         for (; i + 32 <= dim; i += 32) {
             __m256 va0 = _mm256_loadu_ps(a + i);
             __m256 vb0 = _mm256_loadu_ps(b + i);
             __m256 d0  = _mm256_sub_ps(va0, vb0);
             sum0 = _mm256_fmadd_ps(d0, d0, sum0);

             __m256 va1 = _mm256_loadu_ps(a + i + 8);
             __m256 vb1 = _mm256_loadu_ps(b + i + 8);
             __m256 d1  = _mm256_sub_ps(va1, vb1);
             sum1 = _mm256_fmadd_ps(d1, d1, sum1);

             __m256 va2 = _mm256_loadu_ps(a + i + 16);
             __m256 vb2 = _mm256_loadu_ps(b + i + 16);
             __m256 d2  = _mm256_sub_ps(va2, vb2);
             sum2 = _mm256_fmadd_ps(d2, d2, sum2);

             __m256 va3 = _mm256_loadu_ps(a + i + 24);
             __m256 vb3 = _mm256_loadu_ps(b + i + 24);
             __m256 d3  = _mm256_sub_ps(va3, vb3);
             sum3 = _mm256_fmadd_ps(d3, d3, sum3);
         }

         // Combine the 4 independent accumulators
         __m256 sum = _mm256_add_ps(_mm256_add_ps(sum0, sum1), _mm256_add_ps(sum2, sum3));

         // Remainder 8-element chunks
         for (; i + 8 <= dim; i += 8) {
             __m256 va = _mm256_loadu_ps(a + i);
             __m256 vb = _mm256_loadu_ps(b + i);
             __m256 d  = _mm256_sub_ps(va, vb);
             sum = _mm256_fmadd_ps(d, d, sum);
         }

         // Horizontal reduction
         __m128 lo = _mm256_castps256_ps128(sum);
         __m128 hi = _mm256_extractf128_ps(sum, 1);
         __m128 sum128 = _mm_add_ps(lo, hi);
         sum128 = _mm_hadd_ps(sum128, sum128);
         sum128 = _mm_hadd_ps(sum128, sum128);
         float total = _mm_cvtss_f32(sum128);

         // Scalar cleanup tail
         for (; i < dim; ++i) {
             float d = a[i] - b[i];
             total += d * d;
         }
         return total;
     }
     }
     ```
