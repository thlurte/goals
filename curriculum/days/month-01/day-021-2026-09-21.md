     ```python
     import subprocess
     import json
     import numpy as np

     def run_memory_mountain_sweep():
         # Sweep working set sizes from 4KB to 512MB
         sizes_kb = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 4096, 16384, 65536, 262144, 524288]
         strides = [1, 2, 4, 8, 16, 32, 64]
         results = []

         for sz in sizes_kb:
             for st in strides:
                 cmd = f"./build/benchmarks/bench_distance --benchmark_filter=BM_MemoryMountain/{sz}/{st} --benchmark_format=json"
                 res = subprocess.check_output(cmd, shell=True).decode()
                 results.append({"size_kb": sz, "stride": st, "raw": json.loads(res)})

         with open("research/2026-09-measurement-protocol/notes/week3_memory_mountain.json", "w") as f:
             json.dump(results, f, indent=2)

     if __name__ == "__main__":
         run_memory_mountain_sweep()
     ```
