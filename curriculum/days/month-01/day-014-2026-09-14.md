     ```python
     import subprocess
     import json

     def run_accumulator_sweep():
         # Benchmarking 1, 2, 4, 8, 16 accumulators
         configs = ["1acc", "2acc", "4acc", "8acc", "16acc"]
         dims = [64, 128, 768, 1536]
         records = []
         for cfg in configs:
             for d in dims:
                 cmd = f"perf stat -x, -e cycles,instructions ./build/benchmarks/bench_distance --benchmark_filter=BM_L2_{cfg}/{d}"
                 res = subprocess.check_output(cmd, stderr=subprocess.STDOUT, shell=True).decode()
                 records.append({"config": cfg, "dim": d, "perf": res})
         with open("research/2026-09-measurement-protocol/notes/week2_accumulators.json", "w") as f:
             json.dump(records, f, indent=2)

     if __name__ == "__main__":
         run_accumulator_sweep()
     ```
