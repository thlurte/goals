* **Implementation Tasks**:
  1. **Paged KV Cache Manager & Decoder in Pure PyTorch** in `src/paged_kv_engine.py`:
     ```python
     import torch
     import torch.nn.functional as F
     from typing import List, Dict, Tuple

     class PagedKVCacheEngine:
         def __init__(self, num_blocks: int, block_size: int, num_heads_kv: int, head_dim: int, device: str = "cuda"):
             self.block_size = block_size
             self.num_heads_kv = num_heads_kv
             self.head_dim = head_dim
             self.device = device

             # Monolithic physical memory pools
             self.k_pool = torch.zeros((num_blocks, block_size, num_heads_kv, head_dim), dtype=torch.float16, device=device)
             self.v_pool = torch.zeros((num_blocks, block_size, num_heads_kv, head_dim), dtype=torch.float16, device=device)
             self.free_blocks = list(range(num_blocks))
             self.block_tables: Dict[int, List[int]] = {}

         def allocate(self, req_id: int, num_tokens: int):
             num_blocks_needed = (num_tokens + self.block_size - 1) // self.block_size
             table = []
             for _ in range(num_blocks_needed):
                 assert len(self.free_blocks) > 0, "Out of VRAM blocks"
                 table.append(self.free_blocks.pop())
             self.block_tables[req_id] = table

         def append_token(self, req_id: int, token_idx: int, k_vec: torch.Tensor, v_vec: torch.Tensor):
             logical_block = token_idx // self.block_size
             offset = token_idx % self.block_size
             if logical_block >= len(self.block_tables[req_id]):
                 assert len(self.free_blocks) > 0, "Out of VRAM blocks"
                 self.block_tables[req_id].append(self.free_blocks.pop())

             phys_block = self.block_tables[req_id][logical_block]
             self.k_pool[phys_block, offset] = k_vec
             self.v_pool[phys_block, offset] = v_vec
     ```
