  1. **Implement Paged KV Cache Manager** in `include/secan/gpu/paged_kv_cache.cuh`:
     ```cuda
     #pragma once
     #include <cuda_fp16.h>
     #include <cuda_runtime.h>
     #include <vector>
     #include <cstdint>

     namespace secan {
     constexpr int BLOCK_SIZE = 16;

     struct PagedKVCache {
         half* k_pool; // [num_blocks, BLOCK_SIZE, H_kv, D]
         half* v_pool; // [num_blocks, BLOCK_SIZE, H_kv, D]
         int num_total_blocks;
         std::vector<int> free_blocks;

         void allocate_block(int req_id, std::vector<int>& block_table) {
             if (free_blocks.empty()) return;
             int phys_block = free_blocks.back();
             free_blocks.pop_back();
             block_table.push_back(phys_block);
         }
     };
     }
     ```

     * In `benchmarks/bench_paged_kv.cu`, compare VRAM utilization between Paged KV cache vs naive `std::vector` contiguous reallocation under 100 concurrent streams; assert **$>4.2\times$ memory footprint reduction**.
