  1. **Multi-GPU ColPali Search Simulator in Pure PyTorch** in `src/multigpu_colpali.py`:
     ```python
     import torch
     import torch.nn.functional as F
     from typing import List, Tuple

     class MultiGpuColPaliSimulator:
         def __init__(self, num_devices: int = 2, dim: int = 128):
             self.num_devices = num_devices
             self.dim = dim
             self.shards: List[torch.Tensor] = [] # Sharded document tokens

         def shard_documents(self, all_docs: torch.Tensor):
             # all_docs: [N, num_d_tokens, D]
             chunk_size = (all_docs.shape[0] + self.num_devices - 1) // self.num_devices
             self.shards = [all_docs[i*chunk_size : (i+1)*chunk_size] for i in range(self.num_devices)]

         def search(self, query: torch.Tensor, top_k: int = 10) -> Tuple[torch.Tensor, torch.Tensor]:
             # query: [num_q_tokens, D]
             local_scores = []
             local_indices = []

             for dev_id, shard in enumerate(self.shards):
                 # Compute MaxSim: sum_i max_j (Q_i . D_j)
                 # [num_q_tokens, D] x [num_docs, D, num_d_tokens] -> [num_docs, num_q_tokens, num_d_tokens]
                 sims = torch.einsum('qd,ndm->nqm', query, shard.transpose(1, 2))
                 max_sims = torch.max(sims, dim=-1).values # [num_docs, num_q_tokens]
                 scores = torch.sum(max_sims, dim=-1)     # [num_docs]

                 top_scores, top_idx = torch.topk(scores, min(top_k, len(scores)))
                 local_scores.append(top_scores)
                 local_indices.append(top_idx + dev_id * len(shard))

             # Ring AllGather & global top-k reduction
             all_scores = torch.cat(local_scores)
             all_indices = torch.cat(local_indices)
             final_scores, global_topk = torch.topk(all_scores, top_k)
             return final_scores, all_indices[global_topk]
     ```
