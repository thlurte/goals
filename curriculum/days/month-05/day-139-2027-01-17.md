* **Implementation Tasks**:
  1. **Distributed Multi-GPU Vector Retriever in PyTorch** in `src/multi_gpu_retriever.py`:
     ```python
     import torch
     import torch.distributed as dist
     from typing import Tuple

     class MultiGpuVectorSearcher:
         def __init__(self, local_vectors: torch.Tensor, rank: int, world_size: int):
             self.rank = rank
             self.world_size = world_size
             self.local_vectors = local_vectors.to(f"cuda:{rank}") # [N_local, D]

         def search_batch(self, queries: torch.Tensor, top_k: int) -> Tuple[torch.Tensor, torch.Tensor]:
             """
             Search queries against local shard and all-gather candidates across all GPUs.
             """
             device = self.local_vectors.device
             queries = queries.to(device)

             # 1. Local dot product GEMM
             local_scores = torch.matmul(queries, self.local_vectors.T) # [B, N_local]
             local_topk_scores, local_topk_indices = torch.topk(local_scores, k=top_k, dim=-1)

             # Map local indices to global IDs
             global_topk_indices = local_topk_indices + (self.rank * self.local_vectors.shape[0])

             if self.world_size == 1:
                 return local_topk_scores, global_topk_indices

             # 2. NCCL AllGather candidates across ranks
             gathered_scores = [torch.zeros_like(local_topk_scores) for _ in range(self.world_size)]
             gathered_indices = [torch.zeros_like(global_topk_indices) for _ in range(self.world_size)]

             dist.all_gather(gathered_scores, local_topk_scores)
             dist.all_gather(gathered_indices, global_topk_indices)

             # 3. Final top-k reduction on all gathered candidates
             all_scores = torch.cat(gathered_scores, dim=-1) # [B, world_size * K]
             all_indices = torch.cat(gathered_indices, dim=-1)

             final_scores, best_k_pos = torch.topk(all_scores, k=top_k, dim=-1)
             final_indices = torch.gather(all_indices, dim=-1, index=best_k_pos)

             return final_scores, final_indices
     ```
