* **Implementation Tasks**:
  1. **Metropolis-Hastings MCMC Sampler in Pure PyTorch** in `src/mcmc_sampler.py`:
     ```python
     import torch
     from typing import Callable, Tuple

     def metropolis_hastings_sample(
         target_log_prob: Callable[[torch.Tensor], torch.Tensor],
         init_state: torch.Tensor,
         num_samples: int,
         proposal_std: float = 0.5
     ) -> Tuple[torch.Tensor, float]:
         """
         Metropolis-Hastings MCMC sampler for continuous multidimensional target distributions.
         """
         dim = init_state.shape[0]
         samples = torch.zeros((num_samples, dim), device=init_state.device)
         curr_state = init_state.clone()
         curr_log_prob = target_log_prob(curr_state)
         accepted_count = 0

         for i in range(num_samples):
             # Propose candidate via symmetric Gaussian random walk
             proposal = curr_state + torch.randn_like(curr_state) * proposal_std
             prop_log_prob = target_log_prob(proposal)

             # Acceptance probability in log-space
             log_alpha = prop_log_prob - curr_log_prob
             if torch.log(torch.rand(1, device=init_state.device)) < log_alpha:
                 curr_state = proposal
                 curr_log_prob = prop_log_prob
                 accepted_count += 1

             samples[i] = curr_state

         acceptance_rate = accepted_count / num_samples
         return samples, acceptance_rate
     ```
