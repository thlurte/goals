  1. **PagedAttention Autoregressive Generator in Pure PyTorch** in `src/paged_generator.py`:
     ```python
     import torch
     import math
     from typing import List, Dict

     class PagedKVCache:
         def __init__(self, num_blocks: int, block_size: int, num_heads: int, head_dim: int, device="cuda"):
             self.block_size = block_size
             self.k_cache = torch.zeros((num_blocks, block_size, num_heads, head_dim), device=device)
             self.v_cache = torch.zeros((num_blocks, block_size, num_heads, head_dim), device=device)
             self.free_blocks = list(range(num_blocks))

         def allocate(self) -> int:
             return self.free_blocks.pop(0)

         def free(self, block_id: int):
             self.free_blocks.append(block_id)

     def paged_decode_attention(
         q: torch.Tensor,              # [B, 1, num_heads, head_dim]
         block_tables: List[List[int]],# [B, num_allocated_blocks]
         seq_lens: List[int],          # [B]
         kv_cache: PagedKVCache
     ) -> torch.Tensor:
         """
         Paged decoding attention in PyTorch resolving physical blocks via block_tables.
         """
         B, _, H, d = q.shape
         scale = 1.0 / math.sqrt(d)
         outputs = torch.zeros_like(q)

         for b in range(B):
             seq_len = seq_lens[b]
             table = block_tables[b]
             q_b = q[b, 0] # [H, d]

             # Gather non-contiguous physical K, V tokens
             k_gathered = []
             v_gathered = []
             for phys_block in table:
                 k_gathered.append(kv_cache.k_cache[phys_block]) # [block_size, H, d]
                 v_gathered.append(kv_cache.v_cache[phys_block])

             K_full = torch.cat(k_gathered, dim=0)[:seq_len] # [seq_len, H, d]
             V_full = torch.cat(v_gathered, dim=0)[:seq_len] # [seq_len, H, d]

             # Compute Attention for each head
             # [H, 1, d] x [H, d, seq_len] -> [H, 1, seq_len]
             scores = torch.einsum("hd,shd->hs", q_b, K_full) * scale
             attn_weights = torch.softmax(scores, dim=-1) # [H, seq_len]
             out_b = torch.einsum("hs,shd->hd", attn_weights, V_full) # [H, d]

             outputs[b, 0] = out_b

         return outputs
     ```
