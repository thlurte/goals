     * Across $G$ devices, host manages multiple CUDA contexts via `cudaSetDevice(dev_id)`.
     * NVLink interconnect provides direct GPU-to-GPU memory access at **$600\text{ GB/s}$–$900\text{ GB/s}$**, bypassing host RAM and PCIe bottlenecks.
  2. **Data-Parallel Sharding for Vector Search**:
     * Dataset of $N = 40\text{M}$ vectors is split into $G = 4$ contiguous shards of $10\text{M}$ vectors each.
     * Queries are broadcast to all GPUs simultaneously, executing in parallel with zero inter-GPU synchronization during the scan phase.

     #include "secan/gpu/gpu_ivf_fastscan.cuh"
     #include <vector>

     #include <thread>

     namespace secan {
     class MultiGpuIndex {
     private:
         int num_gpus{1};
         std::vector<std::unique_ptr<GpuIvfFastScanIndex>> gpu_shards;
         std::vector<cudaStream_t> streams;

     public:
         MultiGpuIndex(int n_gpus, size_t total_vectors, size_t dim, size_t num_subspaces)
             : num_gpus(n_gpus), streams(n_gpus) {
             size_t vecs_per_gpu = total_vectors / n_gpus;

             for (int g = 0; g < n_gpus; ++g) {
                 CUDA_CHECK(cudaSetDevice(g));
                 CUDA_CHECK(cudaStreamCreate(&streams[g]));
                 gpu_shards.push_back(
                     std::make_unique<GpuIvfFastScanIndex>(4096, vecs_per_gpu, dim, num_subspaces)
                 );
             }
         }

         ~MultiGpuIndex() {
             for (int g = 0; g < num_gpus; ++g) {
                 CUDA_CHECK(cudaSetDevice(g));
                 cudaStreamDestroy(streams[g]);
             }
         }

         void search_distributed(
             const float* h_queries,
             size_t num_queries,
             size_t top_k,
             NeighborPair* h_final_results
         ) {
             // Broadcast query to each GPU -> Launch shard search -> Gather & merge top-k
         }
     };
     }
