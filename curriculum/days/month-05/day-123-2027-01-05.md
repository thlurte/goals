## 📖 05:30 – 06:30 | Systems & Architecture Reading (60 min)
* **Book**: *NVIDIA CUDA C++ Programming Guide: Shared Memory Bank Conflicts & Padding*
* **Core Microarchitectural Concepts**:
  1. **Shared Memory 32-Bank Organization**:
     * Shared memory is divided into 32 banks of 4-byte words. Bank index = `(byte_address / 4) % 32`.
     * If 32 threads in a warp access distinct addresses that map to the same bank, hardware serializes the requests into a 32-way bank conflict stall (32 clock cycles instead of 1).
  2. **Padding Technique for Conflict-Free 2D Access**:
     * Stride $S = 256$ floats aligns with Bank 0 for every row ($256 \equiv 0 \pmod{32}$).
     * By adding a 1-float padding stride: `float s_lut[16][257]`, row $m$ starts at bank `(m * 257) % 32 = m % 32`.
     * Diagonal skewing completely eliminates bank conflicts during vectorized lookups!

     ```python
     import torch
     from typing import Callable, Tuple

     def metropolis_hastings_sample(
         target_log_prob: Callable[[torch.Tensor], torch.Tensor],
         init_state: torch.Tensor,
         num_samples: int,
         proposal_std: float = 0.5
     ) -> Tuple[torch.Tensor, float]:
         """
         Metropolis-Hastings MCMC sampler for continuous multidimensional target distributions.
         """
         dim = init_state.shape[0]
         samples = torch.zeros((num_samples, dim), device=init_state.device)
         curr_state = init_state.clone()
         curr_log_prob = target_log_prob(curr_state)
         accepted_count = 0

         for i in range(num_samples):
             # Propose candidate via symmetric Gaussian random walk
             proposal = curr_state + torch.randn_like(curr_state) * proposal_std
             prop_log_prob = target_log_prob(proposal)

             # Acceptance probability in log-space
             log_alpha = prop_log_prob - curr_log_prob
             if torch.log(torch.rand(1, device=init_state.device)) < log_alpha:
                 curr_state = proposal
                 curr_log_prob = prop_log_prob
                 accepted_count += 1

             samples[i] = curr_state

         acceptance_rate = accepted_count / num_samples
         return samples, acceptance_rate
     ```

     #include "secan/gpu/gpu_buffer.cuh"

     namespace secan {
     constexpr int PADDED_PQ_K = 257; // +1 float padding to skew bank indices

     __global__ void gpu_pq_adc_padded_kernel(
         const float* __restrict__ query_lut,     // [16, 256]
         const uint8_t* __restrict__ pq_codes,    // [N, 16]
         float* __restrict__ out_distances,       // [N]
         size_t num_vectors
     ) {
         __shared__ float s_lut[16][PADDED_PQ_K];
         size_t tid = threadIdx.x;

         // Load LUT with stride skew
         for (size_t m = 0; m < 16; ++m) {
             for (size_t k = tid; k < 256; k += blockDim.x) {
                 s_lut[m][k] = query_lut[m * 256 + k];
             }
         }
         __syncthreads();

         size_t vec_idx = blockIdx.x * blockDim.x + threadIdx.x;
         if (vec_idx >= num_vectors) return;

         const uint8_t* code = pq_codes + vec_idx * 16;
         float dist = 0.0f;

         #pragma unroll
         for (int m = 0; m < 16; ++m) {
             dist += s_lut[m][code[m]];
         }

         out_distances[vec_idx] = dist;
     }
     }
