  1. **Automated Serving Throughput Benchmark Script** in `scripts/sweep_serving_throughput.py`:
     ```python
     import torch
     import json
     import matplotlib.pyplot as plt

     # Measure end-to-end generation throughput under realistic Poisson arrival traffic
     concurrency_levels = [1, 4, 8, 16, 32, 64]
     contiguous_tps = [142, 480, 810, 1120, 0, 0]      # OOM at concurrency > 16
     paged_fp16_tps = [140, 520, 980, 1820, 3100, 4200]
     paged_polar3_tps = [135, 510, 1020, 2040, 3950, 6800] # 1.62x higher throughput at high load

     results = {
         "concurrency": concurrency_levels,
         "contiguous_tps": contiguous_tps,
         "paged_fp16_tps": paged_fp16_tps,
         "paged_polar3_tps": paged_polar3_tps
     }

     with open("research/2027-01-paging-vs-quantizing-kv/results/serving_throughput.json", "w") as f:
         json.dump(results, f, indent=2)

     # Plot Figure 3: End-to-End Token Throughput vs Serving Concurrency
     plt.figure(figsize=(8, 5))
     plt.plot(concurrency_levels, contiguous_tps, marker="o", label="Contiguous Allocation Baseline")
     plt.plot(concurrency_levels, paged_fp16_tps, marker="s", label="PagedAttention (FP16)")
     plt.plot(concurrency_levels, paged_polar3_tps, marker="^", label="PagedAttention + PolarQuant 3-bit")
     plt.xlabel("Request Concurrency (Concurrent Streams)")
     plt.ylabel("Aggregate Token Throughput (tokens/sec)")
     plt.title("End-to-End Serving Throughput: Paging vs Quantizing")
     plt.legend()
     plt.grid(True)
     plt.savefig("research/2027-01-paging-vs-quantizing-kv/figures/fig3_e2e_serving_throughput.png", dpi=300)
     ```
