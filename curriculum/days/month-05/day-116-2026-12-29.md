* **Core Microarchitectural Concepts**:
  1. **1-Warp per Query Beam Search Execution Model**:
     * Launching thread blocks with 8 warps (256 threads) allows each block to evaluate 8 queries concurrently with zero cross-query synchronization.
     * Within each warp:
       * 32 threads cooperatively evaluate the distances from the query to all 32 neighbors of the current expanding node.
       * Uses in-register warp shuffle bitonic networks to maintain the top-$L$ frontier queue ($L \le 64$).
  2. **Eliminating Global Memory Allocation in Hot Loop**:
     * Dynamic `std::priority_queue` allocations in global memory on GPU destroy performance.
     * Storing candidate queues and visited bitmasks in registers and shared memory achieves $>100\times$ speedup!

  1. **Pure PyTorch Batched Graph Beam Search** in `src/cagra_pytorch.py`:
     ```python
     import torch
     from typing import Tuple

     def batched_graph_search_torch(
         queries: torch.Tensor,       # [B, d]
         vectors: torch.Tensor,       # [N, d]
         adjacency: torch.Tensor,     # [N, DEGREE] (fixed degree 32)
         entry_point: int = 0,
         search_iters: int = 16,
         beam_size: int = 32
     ) -> Tuple[torch.Tensor, torch.Tensor]:
         """
         Batched graph beam search implementation in PyTorch tensors.
         """
         B, d = queries.shape
         device = queries.device

         curr_nodes = torch.full((B,), entry_point, dtype=torch.long, device=device)
         best_dists = torch.sum((queries - vectors[curr_nodes]) ** 2, dim=-1)

         for it in range(search_iters):
             # Gather neighbors: [B, DEGREE]
             nbr_indices = adjacency[curr_nodes] # [B, 32]
             nbr_vecs = vectors[nbr_indices]     # [B, 32, d]

             # Compute distances to all neighbors: [B, 32]
             q_expanded = queries.unsqueeze(1)   # [B, 1, d]
             dists = torch.sum((q_expanded - nbr_vecs) ** 2, dim=-1)

             min_nbr_dists, min_nbr_idx = torch.min(dists, dim=-1)
             improved = min_nbr_dists < best_dists

             curr_nodes = torch.where(improved, nbr_indices[torch.arange(B), min_nbr_idx], curr_nodes)
             best_dists = torch.where(improved, min_nbr_dists, best_dists)

         return curr_nodes, best_dists
     ```

  1. **Implement Warp-Cooperative GPU Graph Beam Search Kernel** in `include/secan/gpu/cagra_search.cuh`:
     ```cuda
     #pragma once
     #include "secan/gpu/cagra_layout.cuh"
     #include "secan/gpu/topk_select.cuh"
     #include <cuda_runtime.h>

     namespace secan {
     constexpr int BEAM_WIDTH = 32;

     __global__ void cagra_graph_search_kernel(
         const float* __restrict__ queries,     // [B, dim]
         const float* __restrict__ vectors,     // [N, dim]
         const uint32_t* __restrict__ adjacency,// [N, 32]
         uint32_t entry_point,
         NeighborPair* __restrict__ results,    // [B, K]
         size_t num_queries,
         size_t dim,
         size_t search_iterations
     ) {
         size_t warp_id = (blockIdx.x * blockDim.x + threadIdx.x) / 32;
         size_t lane_id = threadIdx.x % 32;

         if (warp_id >= num_queries) return;

         const float* q = queries + warp_id * dim;

         // In-register candidate pair for lane
         NeighborPair curr_candidate{1e30f, 0xFFFFFFFF};
         uint32_t curr_node = entry_point;

         for (size_t it = 0; it < search_iterations; ++it) {
             // Lane reads its neighbor from the 32-degree adjacency list
             uint32_t nbr_id = adjacency[curr_node * CAGRA_DEGREE + lane_id];

             if (nbr_id != 0xFFFFFFFF) {
                 const float* nbr_vec = vectors + nbr_id * dim;
                 float dist = 0.0f;
                 for (size_t k = 0; k < dim; ++k) {
                     float diff = q[k] - nbr_vec[k];
                     dist += diff * diff;
                 }
                 curr_candidate.dist = dist;
                 curr_candidate.id = nbr_id;
             }

             // Sort 32 candidates in registers using Bitonic sort
             warp_bitonic_sort_32(curr_candidate);

             // Broadcast best unvisited node to lane 0
             curr_node = __shfl_sync(0xffffffff, curr_candidate.id, 0);
         }

         if (lane_id == 0) {
             results[warp_id] = curr_candidate;
         }
     }
     }
     ```
