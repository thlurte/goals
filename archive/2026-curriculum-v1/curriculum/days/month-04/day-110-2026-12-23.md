* **Book**: *NVIDIA CUDA C++ Programming Guide: CUDA Streams & Async Pipelining*
* **Core Microarchitectural Concepts**:
  1. **3-Stage Hardware Stream Concurrency**:
     * Modern GPUs possess independent Hardware DMA Copy Engines and SM Compute Pipelines.
     * By partitioning queries into chunks $C_0, C_1, C_2, \dots$ across 3 CUDA streams:
       * **Stream 1**: Copies Query Chunk $k+1$ from Host to Device VRAM (`cudaMemcpyAsync`).
       * **Stream 2**: Executes GPU IVF multi-probe search kernel for Chunk $k$ on SMs.
       * **Stream 3**: Copies Top-$k$ results for Chunk $k-1$ from Device to Host DRAM.
     * Eliminates PCIe transfer latency from the critical search path!

     namespace secan {
     class PipelinedGpuIvfEngine {
     private:
         static constexpr int NUM_STREAMS = 3;
         cudaStream_t streams[NUM_STREAMS];
         cudaEvent_t events[NUM_STREAMS];

     public:
         PipelinedGpuIvfEngine() {
             for (int i = 0; i < NUM_STREAMS; ++i) {
                 CUDA_CHECK(cudaStreamCreate(&streams[i]));
                 CUDA_CHECK(cudaEventCreate(&events[i]));
             }
         }

         ~PipelinedGpuIvfEngine() {
             for (int i = 0; i < NUM_STREAMS; ++i) {
                 cudaStreamDestroy(streams[i]);
                 cudaEventDestroy(events[i]);
             }
         }

         void search_streamed(
             const float* h_queries,
             size_t num_queries,
             size_t chunk_size,
             NeighborPair* h_results
         ) {
             size_t num_chunks = (num_queries + chunk_size - 1) / chunk_size;
             for (size_t c = 0; c < num_chunks; ++c) {
                 int stream_idx = c % NUM_STREAMS;
                 // Issue async upload -> async kernel -> async download in round-robin streams
             }
             CUDA_CHECK(cudaDeviceSynchronize());
         }
     };
     }
