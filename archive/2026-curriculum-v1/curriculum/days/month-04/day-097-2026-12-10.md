* **Core Microarchitectural Concepts**:
  1. **Constant Memory (`__constant__`) & Warp Broadcast**:
     * $64\text{ KB}$ read-only memory space with dedicated constant cache.
     * When all threads in a warp read query vector floats simultaneously, the constant cache serves the load in **1 clock cycle** via hardware broadcast.
  2. **Fused SFU Instructions (`__rsqrtf`)**:
     * Cosine similarity requires normalizations: $\frac{\mathbf{q} \cdot \mathbf{d}}{\|\mathbf{q}\| \|\mathbf{d}\|}$.
     * Using NVIDIA Special Function Unit (SFU) reciprocal square root intrinsic `__rsqrtf(x)` evaluates $1/\sqrt{x}$ in **1 clock cycle** (vs 14 cycles for `1.0f / sqrtf(x)`).

     ```cuda
     #include "secan/gpu/gpu_buffer.cuh"
     #include "secan/gpu/warp_reduction.cuh"
     #include <cuda_runtime.h>

     namespace secan {
     // Fused single-pass Cosine Similarity: computes dot product, ||q||^2, and ||d||^2 simultaneously
     __global__ void cosine_similarity_fused_kernel(
         const float* __restrict__ queries, // [B, dim]
         const float* __restrict__ database,// [N, dim]
         float* __restrict__ similarities,  // [B, N]
         size_t num_queries,
         size_t num_database,
         size_t dim
     ) {
         size_t global_warp_id = (blockIdx.x * blockDim.x + threadIdx.x) / 32;
         size_t lane_id = threadIdx.x % 32;

         size_t total_pairs = num_queries * num_database;
         if (global_warp_id >= total_pairs) return;

         size_t q_idx = global_warp_id / num_database;
         size_t db_idx = global_warp_id % num_database;

         const float* q = queries + q_idx * dim;
         const float* d = database + db_idx * dim;

         float dot = 0.0f;
         float q_norm_sq = 0.0f;
         float d_norm_sq = 0.0f;

         for (size_t k = lane_id; k < dim; k += 32) {
             float q_val = q[k];
             float d_val = d[k];
             dot += q_val * d_val;
             q_norm_sq += q_val * q_val;
             d_norm_sq += d_val * d_val;
         }

         dot = warp_reduce_sum(dot);
         q_norm_sq = warp_reduce_sum(q_norm_sq);
         d_norm_sq = warp_reduce_sum(d_norm_sq);

         if (lane_id == 0) {
             float inv_norm = __rsqrtf(q_norm_sq * d_norm_sq + 1e-12f);
             similarities[q_idx * num_database + db_idx] = dot * inv_norm;
         }
     }
     }
     ```
